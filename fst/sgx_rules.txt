define Vowel [ a | e | i | o | u | ü | "_" | {a_} ] ;
define LowVowel [ a | {a_} ] ;
define HighVowel [ i | ü | u ] ;
define FrontVowel [ a | e | i ] ;
define BackVowel [ u | ü | o ] ;
define Consonant [ p | {p'} | m | s | c h | h | t | {t'} | k | {k'} | n | ł | l | k w | k y | x | x w | x y | y | ẅ | ' | "_" ] ;
define GlotCons [ {p'} | {t'} | {k'} | {g'} | {'m} | {'n} | {'l} | {'w} | {'y} | {'ẅ} | {ts'} ] ;
define Glottal [ ' ] ;
define CLetter [ b | d | dz | g | g̱ | {g'} | g w | g y | "_" ] ;
define Nasal [ m | n | {'m} | {'n} ] ;
define Sonorant [ m | n | l | w | y | {'m} | {'n} | {'l} | {'w} | {'y} | {'ẅ} ] ;
define Sibilant [ s | c h | {ts} | {ts'} ] ;
define InitialCons [ b | c h | d | d z | g | g w | g y | h | k | {k'} | k w | k y | l | {'l} | ł | m | n | {'n} | p | {p'} | s | t | {t'} | {ts} | {ts'} | w | {'w} | ẅ | {'ẅ} | x | x w | x y | y | {'y} ] ;

!--optional spirantisation of back stop {k_} > x
define SpirantBack  {k_} (->) x || _ "^" .#. ;

!-- spirantisation by adjacency {k_} > {x_} by Sibilant or {ł}
define SpirantAdj   {k_} (->) {x_} || _ "^" [ Sibilant | ł ] ;

!--epenthesis, stop cluster tkt/łkt > tgit/łgit
define StopClusterEp    [..] -> i || {tk} "^" _ t .#. ,,
                        [..] -> i || {łk} "^" _ t .#. ;

!-- ts -> dz alternation, specialised for nts'i'its versus normal
define DZVoicing	{ts} -> [ {ts} | {dz} ] || [ HighVowel | Glottal ] _ "^" [ Vowel | Nasal ],,
                    {ts} -> {dz} || LowVowel _ "^" [ Vowel | Nasal ] ;

!--optional epenthesis, low front vowel after glottal, before stop
define GlottalStopEp    [..] (->) [ LowVowel & FrontVowel ] || ' "^" _ [ t | {g_} ] ;

!-- delete DoubleDentalStop, double t becomes single t 
define DoubleDentalStop t -> 0 || t "^" _ .#. ; 

!-- delete DoubleNasalStop, double mm and nn become single m / n 
define DoubleNasalStop  [..] -> i || m _"^" m .#. ,,
                        n -> 0 || n "^" _ ;

!-- glottal movement, Interrupted vowels become variable stems
define GlottalMove1 {a'a} -> [ {a'a} | {aa'} | {a'} ] || _ t .o. {a'a} -> [ {a'a} | {a'} ] || _ [ Sonorant | {x_} | {k_} ] ;
define GlottalMove2 {i'i} -> [ {i'i} | {ii'} ] || _ [ {ts} ] .o. {i'i} -> [ {i'i} | {i'}] || _ Sonorant ;  
define GlottalMove3 {o'o} -> [ {o'o} | {o'} ] || _ [ Nasal | .#. ]; 

!-- Hardening in post-interrupted vowel {x_} (->) {k_} post shortening, root specific to hana'ax_ >> hana'k_
define PostGM1Hardening {x_} -> {k_} || {a'} _ ;

!-- Long vowel shortening, Some interrupted vowels optionally undergo shortening post-movement {ii'} -> {i'}
define LongVowelShorten {ii'} -> [ {i'} | {ii'} ] || _ ;

!-- voicing rule, vowel adjacency
define VowelVoicing k -> g || _ "^" Vowel ,,
                    p -> b || _ "^" Vowel ,,
                    t -> d || _ "^" Vowel ,,
                    {k_} -> {g_} || _ "^" Vowel ,,
                    {ky} -> {gy} || _ "^" Vowel ,,
                    {kw} -> {gw} || _ "^" Vowel ,,
                    {x_} -> {g_} || _ "^" Vowel ;

!-- cluster devoicing, g_ becomes x_ if in cluster {sx_}
define ClusterDevoicing {g_} -> {x_} || s _ "^" [ Vowel | Nasal ] .#. ;

!-- VelarStop voicing rule nasal adjacency
define VelarNasalVoicing    {kw} -> {gw} || _ "^" [ n | m ] ,,
                            {ky} -> {gy} || _ "^" [ n | m ] ,,
                            {k_} -> {g_} || _ "^" [ n | m ] ;

!-- voicing rule, nasal adjacency
define NasalVoicing k -> g || _ "^" [ n | m ] ,,
                    p -> b || _ "^" [ n | m ] ,,
                    t -> d || _ "^" [ n | m ] ,,
                    {ky} -> {gy} || _ "^" [ n | m ] ,,
                    {x_} -> {g_} || _ "^" [ n | m ] ;

!-- epenthesis, voiced uvular trigger diphthongization  
define VoicedUvularEp   [..] -> a || {g_} _ "^" [ i | {it}] ;

!-- epenthesis, unvoided uvular trigger diphthongization only -i, not -it
define UnVoicedUvularEp [..] -> a || {x_} _ "^" i   ;

!-- glide insertion, postuvular stop preceding rounded vowel
define PostUvularStopGI [..] -> y || {g_} _ "^" u ;

!--when Series II -sm follows a sibilant affricate/fricative, insert a Vowel
define SimHiatus [..] -> i || Sibilant "^" _ {sm} ;

!--delete a connective clitic following an identical phoneme
define DoubleFricative  s -> 0 || Sibilant "^" _ ,,
                        ł -> 0 || ł "^" _ ;

!--delete a connective clitic following lł
define LBarredL         l "^" ł (->) ł ||  _ .#. ;

!--delete a SeriesII -i clitic following [ i | ii ]
define DoubleIDeletion  i -> 0 || _ "^" i .#.  ;

!--glide insertion 'y' between ' Vowel and i/u, Vowel and i/u, i and u, h and u
define GlideInsert  [..] -> y || ' Vowel _ "^" i ,,
                    [..] -> y || ' Vowel _ "^" u ,,
                    [..] -> y || [ u u | ü ü | a a | {a_} | e e | o o | a | o ] _ "^" [ i | u ] ,,
                    [..] -> y || i _ "^" u  ,,
                    [..] -> y || h _ "^" u  ;

!--glide insertion 'y' between glottal and i/u
define LateYInsert  [..] -> y || o ' _ [ u | i ] .#.   ;

!--optional irregular root rule, mother noh or no H-DoubleIDeletion
define IrrRootRule  h (->) 0 || Nasal o _ .#. ;

!--delete -h in -1SG.II for Vowel H words like sah 'day'
define HDeletionII  h (->) 0 || [ a | {a_} ] _ "^" i .#. ,,
                    h (->) 0 || [ a | {a_} ] _ "^" t .#. .o. [..] (->) i || [ Sibilant a | Sonorant {a_} ] _ "^" t .#.    ;
